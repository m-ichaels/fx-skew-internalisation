cmake_minimum_required(VERSION 3.20)
project(fx_skew_internalisation CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  add_compile_options(/W3 /EHsc /permissive- /Zc:__cplusplus /utf-8)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  add_compile_options(-Wall -Wextra)
endif()
add_library(fxmm INTERFACE)
target_include_directories(fxmm INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(fxmm_cli src/main.cpp)
target_link_libraries(fxmm_cli PRIVATE fxmm)
set_target_properties(fxmm_cli PROPERTIES OUTPUT_NAME fxmm)
add_executable(fxmm_tests tests/tests.cpp)
target_link_libraries(fxmm_tests PRIVATE fxmm)
enable_testing()
add_test(NAME fxmm_tests COMMAND fxmm_tests)
